<script>
(function () {
  const rawPath = decodeURIComponent(location.pathname);
  const tagMatch = rawPath.match(/^\/tag\/([^/]+)\/?$/i);
  if (tagMatch) {
    location.replace('/tag/?q=' + encodeURIComponent(tagMatch[1]));
    return;
  }
  const requested = rawPath.toLowerCase()
    .replace(/\d{4}\/\d{2}\/\d{2}/g, ' ')
    .replace(/[^0-9a-z가-힣]+/g, ' ')
    .trim();
  const reqTokens = new Set(requested.split(/\s+/).filter(x => x.length >= 2));
  const posts = [
    {% for post in site.posts limit:100 %}
    {title: {{ post.title | jsonify }}, url: {{ post.url | relative_url | jsonify }}},
    {% endfor %}
  ];
  let best = null, bestScore = 0;
  posts.forEach(function(p) {
    const text = (p.title || '').toLowerCase().replace(/[^0-9a-z가-힣]+/g, ' ');
    const tokens = new Set(text.split(/\s+/).filter(x => x.length >= 2));
    let hit = 0;
    reqTokens.forEach(function(t) { if (tokens.has(t)) hit++; });
    const score = hit / Math.max(1, Math.min(reqTokens.size, tokens.size));
    if (score > bestScore) { best = p; bestScore = score; }
  });
  const el = document.getElementById('recovery-status');
  if (best && bestScore >= 0.45) {
    el.innerHTML = '유사한 게시물을 찾았습니다: <a href="' + best.url + '">' + best.title + '</a>';
  } else {
    el.textContent = '일치하는 게시물을 찾지 못했습니다. 블로그 홈에서 다시 선택해 주세요.';
  }
})();
</script>
